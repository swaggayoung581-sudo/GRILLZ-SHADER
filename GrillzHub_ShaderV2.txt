local RGB = Color3.fromRGB

local conns = {}
local function track(c)
	table.insert(conns, c)
	return c
end

local function create(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		inst[k] = v
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function corner(parent, r)
	return create("UICorner", { CornerRadius = UDim.new(0, r) }, parent)
end

local function tween(inst, t, props, style, dir)
	local tw = TweenService:Create(
		inst,
		TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
		props
	)
	tw:Play()
	return tw
end

-- Rotating red/black gradients
local spinners = {}
local function redBlackGradient(parent, speed, rot, colA)
	local a = colA or C_RED
	local g = create("UIGradient", {
		Color = ColorSequence.new({
			ColorSequenceKeypoint.new(0, a),
			ColorSequenceKeypoint.new(0.5, RGB(14, 14, 14)),
			ColorSequenceKeypoint.new(1, a),
		}),
		Rotation = rot or 0,
	}, parent)
	table.insert(spinners, { g, speed or 60 })
	return g
end

track(RunService.Heartbeat:Connect(function(dt)
	for i = #spinners, 1, -1 do
		local s = spinners[i]
		if s[1].Parent then
			s[1].Rotation = (s[1].Rotation + dt * s[2]) % 360
		else
			table.remove(spinners, i)
		end
	end
end))

local function preset(name, c1, c2, o)
	o = o or {}
	return {
		name = name,
		c1 = c1,
		c2 = c2,
		particle = o.particle,
		isDefault = o.isDefault,
		grass = o.grass,
		L = {
			Ambient = o.amb or RGB(70, 70, 70),
			OutdoorAmbient = o.out or RGB(120, 120, 120),
			Brightness = o.br or 2,
			ClockTime = o.clock or 14,
			FogColor = o.fog or RGB(190, 200, 215),
			FogStart = o.fs or 0,
			FogEnd = o.fe or 2000,
		},
		A = {
			Color = o.acol or RGB(199, 214, 235),
			Decay = o.adec or RGB(106, 112, 125),
			Density = o.aden or 0.3,
			Haze = o.ahaze or 1,
			Glare = o.aglare or 0,
		},
		B = { Intensity = o.bi or 0.3, Size = o.bs or 24, Threshold = o.bt or 1 },
		C = {
			Brightness = o.cb or 0,
			Contrast = o.cc or 0.1,
			Saturation = o.cs or 0.1,
			TintColor = o.ct or WHITE,
		},
		S = { Intensity = o.si or 0.1, Spread = o.ss or 0.5 },
	}
end

local function defaultPreset(c1, c2)
	return preset("Default", c1, c2, { isDefault = true })
end

local function colorPreset(name, col, lite, o)
	o = o or {}
	local k = lite and 0.35 or 0.7
	local c1, c2
	if lite then
		c1 = col:Lerp(WHITE, 0.35)
		c2 = col:Lerp(BLACK, 0.3)
	else
		c1 = col
		c2 = col:Lerp(BLACK, 0.7)
	end
	return preset(name .. (lite and " Lite" or ""), c1, c2, {
		clock = 14,
		br = lite and 2.1 or 2.4,
		amb = RGB(60, 60, 60):Lerp(col, k * 0.35),
		out = RGB(115, 115, 115):Lerp(col, k * 0.4),
		fog = RGB(200, 205, 215):Lerp(col, k * 0.5),
		fe = lite and 2500 or 1800,
		acol = WHITE:Lerp(col, k * 0.8),
		adec = RGB(100, 100, 110):Lerp(col, k * 0.6),
		aden = lite and 0.28 or 0.35,
		ahaze = lite and 1 or 1.6,
		bi = lite and 0.25 or 0.6,
		bs = 30,
		bt = 0.9,
		cb = o.cb or 0,
		cc = lite and 0.08 or 0.2,
		cs = (lite and o.csl) or ((not lite) and o.cs) or (lite and 0.1 or 0.35),
		ct = WHITE:Lerp(col, lite and 0.3 or 0.55),
		si = lite and 0.1 or 0.25,
		ss = 0.6,
	})
end

----------------------------------------------------------------------
-- Grass + weather particles

local grassOn = true

-- Real grass blades spawned on the ground around you (works on Baseplate, parts and terrain)
local GRASS_RADIUS = 26 -- studs around you (lower = less lag)
local GRASS_CELL = 2.0 -- spacing between tufts (lower = denser)
local GRASS_PER_CELL = 4 -- blades per tuft (higher = thicker)
local GRASS_BUDGET = 120 -- new tufts placed per step
local DEFAULT_GRASS = { RGB(58, 140, 52), RGB(110, 190, 70) }

local grassFolder, grassConn
local grassCells = {}
local grassFree = {}
local grassShade = setmetatable({}, { __mode = "k" })
local grassPalette = DEFAULT_GRASS
local grassAcc = 0
local grassRng = Random.new()
local grassParams = RaycastParams.new()
grassParams.FilterType = Enum.RaycastFilterType.Exclude
grassParams.IgnoreWater = true

local groundOrig = {} -- big ground parts we recolored (restored when grass is off)
local function groundColor()
	return grassPalette[1]:Lerp(BLACK, 0.3)
end

local function grassColor(part)
	return grassPalette[1]:Lerp(grassPalette[2], grassShade[part] or 0.5)
end

local function setGrassPalette(a, b)
	grassPalette = { a or DEFAULT_GRASS[1], b or DEFAULT_GRASS[2] }
	for part in pairs(groundOrig) do
		if part.Parent then
			part.Color = groundColor()
		end
	end
	for _, cell in pairs(grassCells) do
		for _, p in ipairs(cell.parts) do
			p.Color = grassColor(p)
		end
	end
end

local PARK = CFrame.new(0, -5000, 0)

local function makeBlades()
	local group = {}
	for i = 1, GRASS_PER_CELL do
		local p = Instance.new("WedgePart")
		p.Anchored = true
		p.CanCollide = false
		p.CanQuery = false
		p.CanTouch = false
		p.CastShadow = false
		p.Material = Enum.Material.Grass
		p.CFrame = PARK
		p.Parent = grassFolder
		group[i] = p
	end
	return group
end

local function placeBlades(parts, pos)
	local yaw = grassRng:NextNumber(0, math.pi)
	for i, p in ipairs(parts) do
		local h = grassRng:NextNumber(0.9, 1.5)
		local w = grassRng:NextNumber(0.3, 0.55)
		p.Size = Vector3.new(0.08, h, w)
		local off = Vector3.new(grassRng:NextNumber(-1, 1), h / 2 - 0.05, grassRng:NextNumber(-1, 1))
		p.CFrame = CFrame.new(pos + off)
			* CFrame.Angles(
				math.rad(grassRng:NextNumber(-8, 8)),
				yaw + (i - 1) * math.pi / 4,
				math.rad(grassRng:NextNumber(-8, 8))
			)
		grassShade[p] = grassRng:NextNumber(0, 1)
		p.Color = grassColor(p)
	end
end

-- turns big flat ground parts (Baseplate etc.) green so the grass looks like one field
local function tintGround(inst)
	if groundOrig[inst] or inst:IsA("Terrain") or not inst:IsA("BasePart") then
		return
	end
	if inst.Size.X < 30 or inst.Size.Z < 30 then
		return
	end
	groundOrig[inst] = { inst.Color, inst.Material }
	inst.Color = groundColor()
	inst.Material = Enum.Material.Grass
end

local function grassStep()
	if not grassFolder then
		return
	end
	local char = LocalPlayer.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	local camera = Workspace.CurrentCamera
	local center = (root and root.Position) or (camera and camera.CFrame.Position)
	if not center then
		return
	end

	local filter = { grassFolder }
	if char then
		table.insert(filter, char)
	end
	local weather = Workspace:FindFirstChild("GH_Weather")
	if weather then
		table.insert(filter, weather)
	end
	grassParams.FilterDescendantsInstances = filter

	local cx = math.floor(center.X / GRASS_CELL)
	local cz = math.floor(center.Z / GRASS_CELL)
	local R = math.ceil(GRASS_RADIUS / GRASS_CELL)

	-- recycle tufts that are now too far away
	for key, cell in pairs(grassCells) do
		local dx, dz = cell.x - cx, cell.z - cz
		if dx * dx + dz * dz > (R + 1) * (R + 1) then
			for _, p in ipairs(cell.parts) do
				p.CFrame = PARK
			end
			table.insert(grassFree, cell.parts)
			grassCells[key] = nil
		end
	end

	-- fill new tufts coming into range
	local made = 0
	for dx = -R, R do
		for dz = -R, R do
			if dx * dx + dz * dz <= R * R then
				local x, z = cx + dx, cz + dz
				local key = x * 131072 + z
				if not grassCells[key] then
					if made >= GRASS_BUDGET then
						return
					end
					made += 1
					local parts = table.remove(grassFree) or makeBlades()
					local wx = (x + grassRng:NextNumber(0.2, 0.8)) * GRASS_CELL
					local wz = (z + grassRng:NextNumber(0.2, 0.8)) * GRASS_CELL
					local hit = Workspace:Raycast(Vector3.new(wx, center.Y + 12, wz), Vector3.new(0, -70, 0), grassParams)
					local ok = false
					if hit and hit.Normal.Y > 0.75 and hit.Instance.Transparency < 0.9 then
						local m = hit.Instance:FindFirstAncestorOfClass("Model")
						if not (m and m:FindFirstChildOfClass("Humanoid")) then
							ok = true
							tintGround(hit.Instance)
							placeBlades(parts, hit.Position)
						end
					end
					if not ok then
						for _, p in ipairs(parts) do
							p.CFrame = PARK
						end
					end
					grassCells[key] = { x = x, z = z, parts = parts }
				end
			end
		end
	end
end

local function startGrass()
	if grassConn then
		return
	end
	if not grassFolder then
		grassFolder = create("Folder", { Name = "GH_Grass" }, Workspace)
	end
	grassAcc = 1
	grassConn = RunService.Heartbeat:Connect(function(dt)
		grassAcc += dt
		if grassAcc >= 0.1 then
			grassAcc = 0
			pcall(grassStep)
		end
	end)
end

local function stopGrass()
	if grassConn then
		grassConn:Disconnect()
		grassConn = nil
	end
	if grassFolder then
		grassFolder:Destroy()
		grassFolder = nil
	end
	for part, o in pairs(groundOrig) do
		if part.Parent then
			part.Color = o[1]
			part.Material = o[2]
		end
	end
	groundOrig = {}
	grassCells = {}
	grassFree = {}
end

local function setGrass(on)
	grassOn = on
	-- also flip the native terrain grass decoration (harmless if locked)
	local terrain = Workspace:FindFirstChildOfClass("Terrain")
	if terrain then
		if sethiddenproperty then
			pcall(sethiddenproperty, terrain, "Decoration", on)
		end
		pcall(function()
			terrain.Decoration = on
		end)
	end
	if on then
		startGrass()
	else
		stopGrass()
	end
end

local activeTweens = {}
local function cancelTweens()
	for _, t in ipairs(activeTweens) do
		t:Cancel()
	end
	table.clear(activeTweens)
end

local function lightTween(inst, props, t)
	local tw = TweenService:Create(inst, TweenInfo.new(t or 0.9, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), props)
	table.insert(activeTweens, tw)
	tw:Play()
end

local function clearEffects()
	for _, v in ipairs(Lighting:GetChildren()) do
		if v:IsA("Atmosphere") or v:IsA("PostEffect") then
			v:Destroy()
		end
	end
end

local function restoreOriginal()
	cancelTweens()
	clearEffects()
	setWeather(nil)
	setGrassPalette()
	for _, e in ipairs(ORIG_EFFECTS) do
		local ok, c = pcall(function()
			return e:Clone()
		end)
		if ok and c then
			c.Parent = Lighting
		end
	end
	lightTween(Lighting, ORIG)
end

local function applyPreset(p)
	if p.isDefault then
		restoreOriginal()
		return
	end
	cancelTweens()
	clearEffects()

	local atm = create("Atmosphere", {
		Name = "GH_Atmosphere", Color = p.A.Color, Decay = p.A.Decay,
		Density = 0, Haze = 0, Glare = 0,
	}, Lighting)
	local bloom = create("BloomEffect", {
		Name = "GH_Bloom", Intensity = 0, Size = p.B.Size, Threshold = p.B.Threshold,
	}, Lighting)
	local cc = create("ColorCorrectionEffect", { Name = "GH_ColorCorrection" }, Lighting)
	local sun = create("SunRaysEffect", { Name = "GH_SunRays", Intensity = 0, Spread = p.S.Spread }, Lighting)

	lightTween(Lighting, p.L)
	lightTween(atm, { Density = p.A.Density, Haze = p.A.Haze, Glare = p.A.Glare })
	lightTween(bloom, { Intensity = p.B.Intensity })
	lightTween(cc, {
		Brightness = p.C.Brightness, Contrast = p.C.Contrast,
		Saturation = p.C.Saturation, TintColor = p.C.TintColor,
	})
	lightTween(sun, { Intensity = p.S.Intensity })

	setWeather(p.particle)
	if p.grass then
		setGrassPalette(p.grass[1], p.grass[2])
	else
		setGrassPalette()
	end
end

local function makeDraggable(handle, target, onClick)
	local dragging, moved, dragStart, startPos = false, false, nil, nil
	track(handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging, moved = true, false
			dragStart, startPos = input.Position, target.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					if dragging and not moved and onClick then
						onClick()
					end
					dragging = false
				end
			end)
		end
	end))
	track(UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			if d.Magnitude > 4 then
				moved = true
			end
			if moved then
				target.Position = UDim2.new(
					startPos.X.Scale, startPos.X.Offset + d.X,
					startPos.Y.Scale, startPos.Y.Offset + d.Y
				)
			end
		end
	end))
end

local function hover(btn, normal, over)
	track(btn.MouseEnter:Connect(function()
		tween(btn, 0.15, { BackgroundColor3 = over })
	end))
	track(btn.MouseLeave:Connect(function()
		tween(btn, 0.15, { BackgroundColor3 = normal })
	end))
end

		Name = cat.id, LayoutOrder = i, Size = UDim2.new(1, 0, 0, 32), BackgroundColor3 = C_CARD,
		Text = cat.label, Font = Enum.Font.GothamMedium, TextSize = 12, TextColor3 = C_SUB,
		TextXAlignment = Enum.TextXAlignment.Left, AutoButtonColor = false, BorderSizePixel = 0,
	}, sidebar)
	corner(btn, 8)
	create("UIPadding", { PaddingLeft = UDim.new(0, 14) }, btn)
	local bar = create("Frame", {
		Position = UDim2.fromOffset(-10, 6), Size = UDim2.fromOffset(3, 20), BackgroundColor3 = C_RED,
		BorderSizePixel = 0, Visible = false,
	}, btn)
	corner(bar, 2)
	sideRefs[cat.id] = { btn = btn, bar = bar }
	track(btn.MouseEnter:Connect(function()
		if currentCat ~= cat.id then
			tween(btn, 0.15, { BackgroundColor3 = C_CARD_HOVER })
		end
	end))
	track(btn.MouseLeave:Connect(function()
		if currentCat ~= cat.id then
			tween(btn, 0.15, { BackgroundColor3 = C_CARD })
		end
	end))
	track(btn.Activated:Connect(function()
		showCategory(cat.id)
	end))
end

----------------------------------------------------------------------
-- Floating GH box (shown when the menu is closed)
----------------------------------------------------------------------
local ghHolder = create("Frame", {
	Name = "GHBox", AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0, 60, 0.5, 0),
	Size = UDim2.fromOffset(48, 48), BackgroundTransparency = 1, Visible = false,
}, gui)
local ghScale = create("UIScale", { Scale = 0.01 }, ghHolder)

local glow = create("Frame", {
	AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(52, 52),
	BackgroundColor3 = C_RED, BackgroundTransparency = 0.55, BorderSizePixel = 0, ZIndex = 1,
}, ghHolder)
corner(glow, 16)
TweenService:Create(
	glow,
	TweenInfo.new(1.1, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
	{ Size = UDim2.fromOffset(70, 70), BackgroundTransparency = 0.92 }
):Play()

local ghBox = create("Frame", {
	AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5), Size = UDim2.fromOffset(46, 46),
	BackgroundColor3 = WHITE, BorderSizePixel = 0, ZIndex = 2,
}, ghHolder)
corner(ghBox, 12)
redBlackGradient(ghBox, 100)
local ghStroke = create("UIStroke", { Thickness = 2, Color = WHITE }, ghBox)
redBlackGradient(ghStroke, -140)
create("TextLabel", {
	Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "GH", Font = Enum.Font.GothamBlack,
	TextSize = 19, TextColor3 = WHITE, TextStrokeTransparency = 0.6, ZIndex = 3,
}, ghBox)

----------------------------------------------------------------------
-- Open / close / minimize
----------------------------------------------------------------------
local isOpen, minimized = true, false

local function openMenu()
	isOpen = true
	tween(ghScale, 0.2, { Scale = 0.01 }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
	task.delay(0.2, function()
		if isOpen then
			ghHolder.Visible = false
		end
	end)
	main.Visible = true
	tween(mainScale, 0.4, { Scale = baseScale }, Enum.EasingStyle.Back)
end

local function closeMenu()
	isOpen = false
	tween(mainScale, 0.25, { Scale = 0.01 }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
	task.delay(0.26, function()
		if not isOpen then
			main.Visible = false
			ghHolder.Visible = true
			tween(ghScale, 0.35, { Scale = 1 }, Enum.EasingStyle.Back)
		end
	end)
end

local function setMinimized(state)
	minimized = state
	if state then
		body.Visible = false
		tween(main, 0.3, { Size = UDim2.fromOffset(W, TITLE_H) })
	else
		tween(main, 0.3, { Size = UDim2.fromOffset(W, H) })
		task.delay(0.15, function()
			if not minimized then
				body.Visible = true
			end
		end)
	end
end

track(btnClose.Activated:Connect(closeMenu))
track(btnMin.Activated:Connect(function()
	setMinimized(not minimized)
end))

makeDraggable(ghBox, ghHolder, openMenu)

track(UserInputService.InputBegan:Connect(function(input, processed)
	if processed then
		return
	end
	if input.KeyCode == Enum.KeyCode.RightShift then
		if isOpen then
			closeMenu()
		else
			openMenu()
		end
	end
end))

setGrass(true) -- grass enabled by default
paintSwitch()
showCategory("shaders") -- default category

task.delay(0.1, function()
	tween(mainScale, 0.6, { Scale = baseScale }, Enum.EasingStyle.Back)
end)
task.delay(0.7, function()
	notify("GRILLZ HUB — SHADER V2", "Loaded  •  Grass enabled  •  RightShift toggles menu", 4)
end)
